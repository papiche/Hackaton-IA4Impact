12URL_FICHE_EPPO = "https://gd.eppo.int/taxon/{code}"
13URL_RECHERCHE_EPPO = "https://gd.eppo.int/search?k={nom}"
14FICHIERS_CODES = ("eppo_seed.csv", "eppo_officiel.csv") # le second, fourni par LPC, complète l'amorce
15SUFFIXES_SANS_SENS = re.compile(r"\b(spp?|ssp|var|cv|l|sensu lato)\b\.?")

18def fold(s: str) -> str:
19 s = unicodedata.normalize("NFKD", s or "")
20 s = "".join(c for c in s if not unicodedata.combining(c)).casefold()
21 return re.sub(r"\s+", " ", s).strip()

24def lien_eppo(code: str) -> str:
25 """Fiche officielle du code sur gd.eppo.int : chaque association reste vérifiable en un clic."""
26 return URL_FICHE_EPPO.format(code=code)

29def nettoyer_nom(nom: str) -> str:
30 """Retire ce qui n'identifie pas l'espèce : parenthèses, « spp. », « var. », auteurs en fin de nom."""
31 nom = re.sub(r"\([^)]*\)", " ", fold(nom))
32 nom = SUFFIXES_SANS_SENS.sub(" ", nom)
33 return re.sub(r"\s+", " ", nom).strip(" .-")

36def singulier(nom: str) -> str:
37 """Pluriels simples : « vignes » -> « vigne », « choux » inchangé grâce à la table des noms."""
38 return " ".join(mot[:-1] if len(mot) > 4 and mot.endswith("s") else mot for mot in nom.split())

41@functools.lru_cache(maxsize=1)
42def table() -> dict[str, list[tuple[str, str]]]:
43 """kind -> [(nom_normalisé, code)]"""
44 t: dict[str, list[tuple[str, str]]] = {"crop": [], "pest": []}
45 for fichier in FICHIERS_CODES:
46 chemin = config.CONFIG / fichier
47 if not chemin.exists():
48 continue
49 with open(chemin, encoding="utf-8") as f:
50 for r in csv.DictReader(f):
51 for n in r["names"].split("|"):
52 t[r["kind"]].append((fold(n), r["code"]))
53 return t

56def _parts(original: str) -> list[str]:
57 return [p.strip() for p in re.split(r"\s+-\s+|,|;|/|\s+et\s+|\s+and\s+", original) if p.strip()]

60def _note(recherche: str, nom_table: str) -> float:
61 """1.0 = même nom (après nettoyage et singulier) ; 0.9 = mêmes mots dans un autre ordre ; sinon ressemblance."""
62 if nom_table == recherche or singulier(nom_table) == singulier(recherche):
63 return 1.0
64 if set(nom_table.split()) == set(recherche.split()):
65 return 0.9
66 note = difflib.SequenceMatcher(None, recherche, nom_table).ratio()
67 if len(nom_table) > 3 and (nom_table in recherche or recherche in nom_table):
68 note = max(note, 0.8)
69 return note

72def candidates(name: str, kind: str, n: int = 5) -> list[tuple[str, str, float]]:
73 f = nettoyer_nom(name)
74 scored = {}
75 for nm, code in table()[kind]:
76 r = _note(f, nettoyer_nom(nm))
77 if code not in scored or r > scored[code][1]:
78 scored[code] = (nm, r)
79 top = sorted(((c, nm, r) for c, (nm, r) in scored.items()), key=lambda x: -x[2])[:n]
80 return [(c, nm, r) for c, nm, r in top if r >= 0.6]

83def _code_par_api(nom: str) -> str | None:
84 """Dernier recours : l'API EPPO, acceptée seulement si elle désigne UN seul code pour ce nom exact."""
85 codes = set(eppo_api.codes_du_nom(nettoyer_nom(nom)) or eppo_api.codes_du_nom(nom))
86 return codes.pop() if len(codes) == 1 else None

89def enrichir_table() -> int:
90 """Complète config/eppo_officiel.csv avec les noms officiels (fr, en, pt, latin) de chaque code connu.
91 Retourne le nombre de noms ajoutés par rapport à l'amorce."""
92 lignes, ajoutes = [], 0
93 with open(config.CONFIG / "eppo_seed.csv", encoding="utf-8") as f:
94 for r in csv.DictReader(f):
95 connus = {fold(n) for n in r["names"].split("|")}
96 nouveaux = [n for n in eppo_api.noms_du_code(r["code"]) if fold(n) not in connus]
97 ajoutes += len(nouveaux)
98 lignes.append({"code": r["code"], "kind": r["kind"], "names": "|".join(r["names"].split("|") + nouveaux)})
99 with open(config.CONFIG / "eppo_officiel.csv", "w", encoding="utf-8", newline="") as f:
100 auteur = csv.DictWriter(f, fieldnames=["code", "kind", "names"])
101 auteur.writeheader(); auteur.writerows(lignes)
102 table.cache_clear()
103 return ajoutes

106def normalize(original: str | None, kind: str) -> dict:
107 """Retourne {"eppo": "A;B"|None, "method": ..., "steps": [...]} (steps = trace pédagogique)."""
108 if not original:
109 return {"eppo": None, "method": "absent", "steps": []}
110 codes, steps = [], []
111 for part in _parts(original):
112 cands = candidates(part, kind)
113 exact = [c for c, nm, r in cands if r == 1.0]
114 if exact:
115 codes.append(exact[0]); steps.append({"part": part, "method": "table", "code": exact[0], "candidates": [c for c, *_ in cands]})
116 continue
117 if not cands:
118 code_officiel = _code_par_api(part)
119 if code_officiel:
120 codes.append(code_officiel)
121 steps.append({"part": part, "method": "api_eppo", "code": code_officiel, "candidates": [code_officiel]})
122 continue
123 steps.append({"part": part, "method": "aucun_candidat", "code": None, "candidates": []})
124 return {"eppo": None, "method": "non_confirme", "steps": steps}
125 listing = "\n".join(f"- {c} ({nm})" for c, nm, _ in cands)
126 res = llm.chat_json(config.PROMPTS.joinpath("eppo.system.md").read_text(),
127 f"Nom à normaliser : « {part} »\nCandidats :\n{listing}", num_predict=60)
128 code = None
129 try:
130 code = EppoChoice.model_validate(json.loads(res["text"] or "{}")).eppo
131 except Exception:
132 pass
133 ok = code in {c for c, *_ in cands} # le LLM ne peut choisir que dans la liste
134 steps.append({"part": part, "method": "llm_parmi_candidats", "code": code if ok else None,
135 "candidates": [c for c, *_ in cands], "llm_raw": res["text"]})
136 if not ok or code is None:
137 return {"eppo": None, "method": "non_confirme", "steps": steps}
138 codes.append(code)
139 return {"eppo": ";".join(dict.fromkeys(codes)) or None, "method": "confirme", "steps": steps}
